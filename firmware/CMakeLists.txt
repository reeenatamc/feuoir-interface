cmake_minimum_required(VERSION 3.13)

set(PICO_BOARD pico CACHE STRING "Placa")

include(pico_sdk_import.cmake)

project(feuoir_firmware C CXX ASM)
set(CMAKE_C_STANDARD 11)

pico_sdk_init()

# SCKI del PCM1808 desde el oscilador externo en vez de GPOUT0 (docs/reloj.md, oscilador externo).
option(FEUOIR_RELOJ_EXTERNO "Tomar SCKI del oscilador externo y dejar GPOUT0 apagado" OFF)

add_executable(feuoir
    main.c
    reloj_maestro.c
)

target_link_libraries(feuoir pico_stdlib hardware_clocks)
target_compile_definitions(feuoir PRIVATE RELOJ_EXTERNO=$<BOOL:${FEUOIR_RELOJ_EXTERNO}>)

pico_add_extra_outputs(feuoir)

# Verificación del reloj maestro, pasos 1 y 2 de docs/reloj.md.
add_executable(verificar_reloj
    verificar_reloj.c
    informe.c
    reloj_maestro.c
)

target_link_libraries(verificar_reloj pico_stdlib hardware_clocks)
target_compile_definitions(verificar_reloj PRIVATE RELOJ_EXTERNO=$<BOOL:${FEUOIR_RELOJ_EXTERNO}>)
pico_enable_stdio_usb(verificar_reloj 1)
pico_enable_stdio_uart(verificar_reloj 0)

pico_add_extra_outputs(verificar_reloj)

# Test UAC2 device: the Mac sees a microphone called "feuoir" and receives a 1 kHz tone generated by the
# Pico itself, no ADC yet. First step of phase 5 (docs/audio-usb.md).
add_executable(tono_usb
    tono_usb.c
    tono_usb_descriptors.c
    reloj_maestro.c
)

target_include_directories(tono_usb PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(tono_usb pico_stdlib hardware_clocks tinyusb_device tinyusb_board m)
target_compile_definitions(tono_usb PRIVATE RELOJ_EXTERNO=$<BOOL:${FEUOIR_RELOJ_EXTERNO}>)

pico_add_extra_outputs(tono_usb)
